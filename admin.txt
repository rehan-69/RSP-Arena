from collections import deque

def print_puzzle(state):
    for row in state:
        print(row)
    print()

def find_zero(state):
    for i in range(3):
        for j in range(3):
            if state[i][j] == 0:
                return i, j

def generate_moves(state):
    moves = []
    x, y = find_zero(state)
    directions = [(-1, 0), (1, 0), (0, -1), (0, 1)]

    for dx, dy in directions:
        nx = x + dx
        ny = y + dy

        if 0 <= nx < 3 and 0 <= ny < 3:
            new_state = [row[:] for row in state]
            new_state[x][y], new_state[nx][ny] = new_state[nx][ny], new_state[x][y]
            moves.append(new_state)

    return moves

def bfs(start, goal):
    queue = deque()
    visited = []

    queue.append((start, []))
    visited.append(start)

    while queue:
        current_state, path = queue.popleft()

        if current_state == goal:
            return path + [current_state]

        for move in generate_moves(current_state):
            if move not in visited:
                visited.append(move)
                queue.append((move, path + [current_state]))

    return None

start = [
    [1, 2, 3],
    [4, 0, 6],
    [7, 5, 8]
]

goal = [
    [1, 2, 3],
    [4, 5, 6],
    [7, 8, 0]
]

solution = bfs(start, goal)

print("Solution Path:\n")
for step in solution:
    print_puzzle(step)

*** 4) uniform_cost_search ***


import heapq

graph = {
    "Mumbai": [("Pune", 150), ("Hyderabad", 710)],
    "Pune": [("Mumbai", 150), ("Bangalore", 840)],
    "Hyderabad": [("Mumbai", 710), ("Bangalore", 570)],
    "Bangalore": [("Pune", 840), ("Hyderabad", 570)]
}

def ucs(start, goal):
    queue = [(0, start, [start])]
    visited = []

    while queue:
        cost, city, path = heapq.heappop(queue)

        if city == goal:
            return cost, path

        if city not in visited:
            visited.append(city)

            for neighbor, distance in graph[city]:
                heapq.heappush(
                    queue,
                    (cost + distance, neighbor, path + [neighbor])
                )

    return None, None

cost, path = ucs("Mumbai", "Bangalore")
print("Least-cost Path")
print("Path:", " -> ".join(path))
print("Total Distance:", cost, "km")

    distance = 0

        for j in range(3):
            value = state[i][j]

def heuristic(state):
    h = 0
    for i in range(3):
        for j in range(3):
            value = state[i][j]
            if value != 0:
                x = (value - 1) // 3
                y = (value - 1) % 3
                h += abs(i - x) + abs(j - y)
    return h

def find_blank(state):
    for i in range(3):
        for j in range(3):
            if state[i][j] == 0:
                return i, j

def neighbors(state):
    x, y = find_blank(state)
    moves = [(1,0),(0,1),(0,-1),(-1,0)]
    result = []

    for dx, dy in moves:
        nx, ny = x + dx, y + dy
        if 0 <= nx < 3 and 0 <= ny < 3:
            new = [row[:] for row in state]
            new[x][y], new[nx][ny] = new[nx][ny], new[x][y]
            result.append(new)

    return result

def astar(start):
    pq = []
    visited = set()

    g = 0
    h = heuristic(start)
    f = g + h

    heappush(pq, (f, g, start))

    while pq:
        f, g, state = heappop(pq)

        if str(state) in visited:

        visited.add(str(state))

        if state == goal:
            print("Goal Reached!")
            return

        for next_state in neighbors(state):
            if str(next_state) not in visited:
                new_g = g + 1
                new_h = heuristic(next_state)
                new_f = new_g + new_h

import heapq

graph = {
    'Arad': [('Zerind', 75), ('Sibiu', 140), ('Timisoara', 118)],
    'Zerind': [('Arad', 75), ('Oradea', 71)],
    'Oradea': [('Zerind', 71), ('Sibiu', 151)],
    'Sibiu': [('Arad', 140), ('Oradea', 151), ('Fagaras', 99), ('Rimnicu Vilcea', 80)],
    'Timisoara': [('Arad', 118), ('Lugoj', 111)],
    'Lugoj': [('Timisoara', 111), ('Mehadia', 70)],
    'Mehadia': [('Lugoj', 70), ('Drobeta', 75)],
    'Drobeta': [('Mehadia', 75), ('Craiova', 120)],
    'Craiova': [('Drobeta', 120), ('Rimnicu Vilcea', 146), ('Pitesti', 138)],
    'Rimnicu Vilcea': [('Sibiu', 80), ('Craiova', 146), ('Pitesti', 97)],
    'Fagaras': [('Sibiu', 99), ('Bucharest', 211)],
    'Pitesti': [('Rimnicu Vilcea', 97), ('Craiova', 138), ('Bucharest', 101)],
    'Bucharest': [('Fagaras', 211), ('Pitesti', 101)]
}

heuristic = {
    'Arad': 366,
    'Zerind': 374,
    'Oradea': 380,
    'Sibiu': 253,
    'Timisoara': 329,
    'Lugoj': 244,
    'Mehadia': 241,
    'Drobeta': 242,
    'Craiova': 160,
    'Rimnicu Vilcea': 193,
    'Fagaras': 176,
    'Pitesti': 100,
    'Bucharest': 0
}

def astar(start, goal):
    queue = []
    heapq.heappush(queue, (heuristic[start], 0, start, [start]))
    visited = set()

    while queue:
        f_cost, g_cost, city, path = heapq.heappop(queue)

        if city == goal:
            return g_cost, path

        if city in visited:
            continue

        visited.add(city)

        for neighbor, distance in graph[city]:
            if neighbor not in visited:
                new_g = g_cost + distance
                new_f = new_g + heuristic[neighbor]
                heapq.heappush(queue, (new_f, new_g, neighbor, path + [neighbor]))

    return None, []

cost, path = astar('Arad', 'Bucharest')
print("Shortest Path using A* Search:")
print("Path:", " -> ".join(path))
print("Total Cost:", cost)

from pgmpy.models import DiscreteBayesianNetwork
from pgmpy.factors.discrete import TabularCPD
from pgmpy.inference import VariableElimination

model = DiscreteBayesianNetwork([('Burglary', 'Alarm')])

cpd_burglary = TabularCPD(
    variable='Burglary',
    variable_card=2,
    values=[[0.999], [0.001]]
)

cpd_alarm = TabularCPD(
    variable='Alarm',
    variable_card=2,
    values=[
        [0.99, 0.05],
        [0.01, 0.95]
    ],
    evidence=['Burglary'],
    evidence_card=[2]
)

model.add_cpds(cpd_burglary, cpd_alarm)

print("Model Valid:", model.check_model())

infer = VariableElimination(model)

result = infer.query(
    variables=['Burglary'],
    evidence={'Alarm': 1}
)

print(result)
